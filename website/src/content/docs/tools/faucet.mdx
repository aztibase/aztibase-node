<script>{`
(function() {
  const RPC = 'https://rpc.aztibase.com';
  const btn = document.getElementById('faucet-btn');
  const input = document.getElementById('faucet-addr');
  const result = document.getElementById('faucet-result');

  function showResult(msg, ok) {
    result.textContent = msg;
    result.className = 'faucet-result ' + (ok ? 'success' : 'error');
    result.style.display = 'block';
  }

  btn.addEventListener('click', async function() {
    const addr = input.value.trim();
    if (!addr.match(/^0x[0-9a-fA-F]{64}$/)) {
      showResult('Invalid address. Must be 0x + 64 hex characters.', false);
      return;
    }

    btn.disabled = true;
    btn.textContent = 'Requesting...';
    result.style.display = 'none';

    try {
      const resp = await fetch(RPC, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          jsonrpc: '2.0',
          method: 'aztb_faucetDrip',
          params: [addr],
          id: 1
        })
      });
      const data = await resp.json();
      if (data.error) {
        showResult('Error: ' + (data.error.message || JSON.stringify(data.error)), false);
      } else {
        showResult('Sent 1,000,000 AZTB\\nTx: ' + (data.result || 'submitted'), true);
      }
    } catch (e) {
      showResult('Network error: ' + e.message, false);
    } finally {
      btn.disabled = false;
      btn.textContent = 'Request Tokens';
    }
  });

  input.addEventListener('keydown', function(e) {
    if (e.key === 'Enter') btn.click();
  });
})();
`}</script>
