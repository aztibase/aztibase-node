## Liquidity

The Aztibase DEX uses an **x*y=k** constant-product AMM (Uniswap V2 pattern). Liquidity providers earn 0.3% on every swap.

### Contracts

| Contract | Purpose |
|----------|---------|
| **WASZTB** | Wrapped native AZTB (ERC20) |
| **AztibaseFactory** | Creates and tracks trading pairs |
| **AztibasePair** | Liquidity pool with LP token |
| **AztibaseRouter** | User-facing swap + liquidity entry point |

### How it works

1. **Wrap** native AZTB into WASZTB (or use existing ERC20 tokens)
2. **Add liquidity** to a pair via the Router — receive LP tokens
3. **Swap** tokens — the Router finds the best path and handles slippage
4. **Remove liquidity** — burn LP tokens, receive underlying tokens

<script>{`
(function() {
  var RPC = 'http://102.209.21.247:9944';
  var SLIPPAGE = 0.005;
  var DECIMALS = 18;

  var CONTRACTS = {
    WASZTB: '0x3990ff2c823c30416e0c8cbb689f4c5dfdcf1132000000000000000000000000',
    tUSDC: '0x6f6a5a8e7d1545955f3814139e6550d03766b4d2000000000000000000000000',
    Factory: '0x34b362ca4a307ea7756a3658861d6cf4f277841f000000000000000000000000',
    Router: '0x909f299b994cd4a3b8c6b6382d589f406ed68248000000000000000000000000',
    PriceFeed: '0x74af2626ed64f9966994a2f0828915b4524d0fae000000000000000000000000'
  };

  function tokenAddr(name) {
    return CONTRACTS[name] || CONTRACTS.WASZTB;
  }

  function pad32(hex) { return hex.replace('0x','').padStart(64,'0'); }
  function toHex256(n) {
    var bn = typeof n === 'bigint' ? n : BigInt(Math.floor(n));
    return bn.toString(16).padStart(64,'0');
  }
  function hexToBigInt(hex) { return BigInt('0x' + hex.replace('0x','')); }

  // Load addresses from deployed config (injected at build or fetched)
  var addrEls = {
    router: document.getElementById('swap-router-addr'),
    factory: document.getElementById('swap-factory-addr'),
    wasztb: document.getElementById('swap-wasztb-addr')
  };

  function rpcCall(method, params) {
    return fetch(RPC, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ jsonrpc: '2.0', method: method, params: params || [], id: 1 })
    }).then(function(r) { return r.json(); }).then(function(j) {
      if (j.error) throw new Error(j.error.message);
      return j.result;
    });
  }

  var amountIn = document.getElementById('swap-amount-in');
  var amountOut = document.getElementById('swap-amount-out');
  var tokenFrom = document.getElementById('swap-token-from');
  var tokenTo = document.getElementById('swap-token-to');
  var flipBtn = document.getElementById('swap-flip');
  var swapBtn = document.getElementById('swap-btn');
  var result = document.getElementById('swap-result');
  var details = document.getElementById('swap-details');

  function showResult(msg, ok) {
    result.textContent = msg;
    result.className = 'swap-result ' + (ok ? 'success' : 'error');
    result.style.display = 'block';
  }

  flipBtn.addEventListener('click', function() {
    var fromVal = tokenFrom.value;
    var toVal = tokenTo.value;
    tokenFrom.value = toVal;
    tokenTo.value = fromVal;
    amountIn.value = '';
    amountOut.value = '';
    details.style.display = 'none';
    swapBtn.textContent = 'Enter an amount';
    swapBtn.disabled = true;
  });

  var quoteTimeout;
  amountIn.addEventListener('input', function() {
    clearTimeout(quoteTimeout);
    var val = parseFloat(amountIn.value);
    if (!val || val <= 0) {
      amountOut.value = '';
      details.style.display = 'none';
      swapBtn.textContent = 'Enter an amount';
      swapBtn.disabled = true;
      return;
    }
    swapBtn.textContent = 'Fetching quote...';
    swapBtn.disabled = true;

    quoteTimeout = setTimeout(function() {
      var amountWei = BigInt(Math.floor(val)) * BigInt(10)**BigInt(DECIMALS);
      var path = [tokenAddr(tokenFrom.value), tokenAddr(tokenTo.value)];
      var calldata = encodeGetAmountsOut(amountWei, path);

      rpcCall('aztb_call', [{ to: CONTRACTS.Router, data: calldata }])
        .then(function(res) {
          var outWei = decodeAmountsOut(res);
          var outFloat = Number(outWei) / (10**DECIMALS);
          amountOut.value = outFloat.toFixed(6);
          var rate = outFloat / val;
          document.getElementById('swap-rate').textContent =
            '1 ' + tokenFrom.value + ' = ' + rate.toFixed(6) + ' ' + tokenTo.value;
          document.getElementById('swap-impact').textContent = '< 0.3%';
          document.getElementById('swap-min-out').textContent =
            (outFloat * (1 - SLIPPAGE)).toFixed(6) + ' ' + tokenTo.value;
          details.style.display = 'block';
          swapBtn.textContent = 'Swap (connect wallet)';
          swapBtn.disabled = false;
        }).catch(function(err) {
          var simulated = val * 0.997;
          amountOut.value = simulated.toFixed(6);
          document.getElementById('swap-rate').textContent =
            '1 ' + tokenFrom.value + ' ~ ' + (0.997).toFixed(6) + ' ' + tokenTo.value + ' (est.)';
          document.getElementById('swap-impact').textContent = '--';
          document.getElementById('swap-min-out').textContent = '--';
          details.style.display = 'block';
          swapBtn.textContent = 'Swap (connect wallet)';
          swapBtn.disabled = false;
        });
    }, 500);
  });

  swapBtn.addEventListener('click', function() {
    showResult('Connect a wallet to swap. Browser wallet extension or aztb_sendTransaction RPC required.', false);
  });

  function encodeGetAmountsOut(amountWei, path) {
    // getAmountsOut(uint256,address[]) selector: 0xd06ca61f
    var sel = 'd06ca61f';
    var amount = toHex256(amountWei);
    var offset = toHex256(64n); // offset to array
    var arrLen = toHex256(BigInt(path.length));
    var addrs = path.map(function(a) { return pad32(a); }).join('');
    return '0x' + sel + amount + offset + arrLen + addrs;
  }
  function decodeAmountsOut(hex) {
    var data = hex.replace('0x','');
    if (data.length < 192) return 0n;
    // Skip offset(32) + length(32) + first amount(32), read second amount
    var secondSlot = data.slice(192, 256);
    return hexToBigInt(secondSlot);
  }

  // Show contract addresses if available
  function updateAddresses() {
    if (CONTRACTS.Router) addrEls.router.textContent = CONTRACTS.Router.slice(0, 10) + '...';
    if (CONTRACTS.Factory) addrEls.factory.textContent = CONTRACTS.Factory.slice(0, 10) + '...';
    if (CONTRACTS.WASZTB) addrEls.wasztb.textContent = CONTRACTS.WASZTB.slice(0, 10) + '...';
  }
  updateAddresses();
})();
`}</script>
